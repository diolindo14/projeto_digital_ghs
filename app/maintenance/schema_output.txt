--- TABLE: utilizadores ---
  [id] int(11)
  [nome_completo] varchar(100)
  [email] varchar(100)
  [senha] varchar(255)
  [tipo] enum('admin','secretaria','professor','aluno')
  [status] enum('ativo','inativo','pendente')
  [token_confirmacao] varchar(100)
  [token_recuperacao] varchar(100)
  [token_expira] datetime
  [ultimo_acesso] datetime
  [ip_registo] varchar(45)
  [data_criacao] timestamp
  [data_atualizacao] timestamp
  [tentativas_login] int(11)
  [bloqueado_ate] datetime
  [codigo_2fa] varchar(10)
  [expiracao_2fa] datetime
--- TABLE: professores ---
  [id] int(11)
  [utilizador_id] int(11)
  [bi] varchar(20)
  [telefone] varchar(20)
  [especialidade] varchar(100)
  [grau_academico] varchar(50)
  [data_contratacao] date
  [foto_perfil] varchar(255)
  [data_criacao] timestamp
  [data_atualizacao] timestamp
--- TABLE: estudantes ---
  [id] int(11)
  [utilizador_id] int(11)

  [data_nascimento] date
  [nacionalidade] varchar(50)
  [sexo] enum('Masculino','Feminino')
  [estado_civil] enum('Solteiro','Casado','Divorciado','Viúvo')
  [telefone] varchar(20)
  [telefone_alternativo] varchar(20)
  [morada] text
  [bairro] varchar(100)
  [cidade] varchar(100)
  [nome_encarregado] varchar(100)
  [telefone_encarregado] varchar(20)
  [escola_proveniencia] varchar(100)
  [ano_conclusao] int(11)
  [media_final] decimal(4,2)
  [foto_perfil] varchar(255)
  [data_criacao] timestamp
  [data_atualizacao] timestamp
